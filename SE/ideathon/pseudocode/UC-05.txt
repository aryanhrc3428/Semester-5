1: procedure UploadDocument(student_id, query_id, file_obj)
2:     // 1. Initial Type and Null Validation
3:     if student_id is null or query_id is null or file_obj is null
4:     then
5:         return error "Validation Failed: Inputs cannot be null"
6:     end if
7:     if typeof(file_obj.name) != "string" or typeof(file_obj.size) != "integer"
8:     then
9:         return error "Validation Failed: Invalid file object structure"
10:    end if
11:    // 2. Explicit Physical File Constraints
12:    max_bytes <- 25 * 1024 * 1024  // 26,214,400 bytes (25MB)
13:    if file_obj.size = 0 or file_obj.size > max_bytes
14:    then
15:        return error "Validation Failed: File size must be between 1 byte and 25MB"
16:    end if
17:    valid_formats <- array of strings {"PDF", "JPG", "JPEG", "PNG"}
18:    file_extension <- toUpper(extractExtension(file_obj.name))
19:    is_valid_format <- false
20:    for i <- 0 to length(valid_formats) - 1
21:    do
22:        if file_extension = valid_formats[i]
23:        then
24:            is_valid_format <- true
25:            break
26:        end if
27:    end for
28:    if is_valid_format = false
29:    then
30:        return error "Validation Failed: Unsupported file format detected"
31:    end if
32:    // 3. Relational Integrity and Ownership Check
33:    try
34:        auth_check <- "SELECT student_id, status FROM D3_QUERIES WHERE query_id = ?"
35:        parent_query <- executeParameterizedQuery(auth_check, [query_id])
36:        if parent_query is empty
37:        then
38:            return error "Database Error: Invalid query reference"
39:        end if
40:        if parent_query.student_id != student_id
41:        then
42:            return error "Security Error: Unauthorized attempt to upload to another user's query"
43:        end if
44:    catch DatabaseException as db_err
45:        logError(db_err)
46:        return error "System Error: Unable to validate query ownership"
47:    end try
48:    // 4. External Cloud Storage Integration with Resilience
49:    storage_key <- null
50:    try
51:        storage_key <- call AWSS3_UploadStream(file_obj.stream, file_obj.name, query_id)
52:        if storage_key is null
53:        then
54:            return error "Cloud Error: S3 returned null storage key"
55:        end if
56:    catch NetworkException as net_err
57:        logError(net_err)
58:        return error "System Error: File upload service temporarily unavailable"
59:    end try
60:    // 5. Secure Database Transaction (Insert Document & Update Query)
61:    new_doc_id <- generateSecureUUID()
62:    current_time <- getSystemTimestamp()
63:    try
64:        beginTransaction()
65:        insert_doc <- "INSERT INTO D4_DOCUMENTS (document_id, query_id, uploaded_by, file_name, file_type, file_size, storage_key, status, uploaded_at) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)"
66:        executeParameterizedInsert(insert_doc, [new_doc_id, query_id, student_id, file_obj.name, file_extension, file_obj.size, storage_key, "Pending Review", current_time])
67:        update_query <- "UPDATE D3_QUERIES SET status = ?, updated_at = ? WHERE query_id = ?"
68:        executeParameterizedUpdate(update_query, ["Document Submitted", current_time, query_id])
69:        commitTransaction()
70:    catch DatabaseException as db_err
71:        rollbackTransaction()
72:        logError(db_err)
73:        // Attempt to clean up orphaned S3 file if DB fails
74:        call AWSS3_Delete(storage_key) 
75:        return error "Database Error: Failed to save document records"
76:    end try
77:    // 6. Fire-and-Forget Asynchronous Notification
78:    try
79:        call DispatchNotificationAsync("Staff", "Document Uploaded", new_doc_id)
80:    catch NetworkException as net_err
81:        logError("Warning: Failed to dispatch staff notification for document " + new_doc_id)
82:    end try
83:    return success "Document securely uploaded and linked", new_doc_id
84: end procedure
